SELECT 	p."Registraèní èíslo projektu",
	p.zadatel,
	p.kodwfstav,
	p.specifcil,
	p.kodspecifcil,
	p.kompetence,
	p.KodPrioritniOsa,
	v."Název výzvy",
	v."Èíslo výzvy",
	hist.CREATE_CAS
--	hist.CREATE_UZIVATEL,
--	hist.StavZ,

FROM		isum.v_bi_umzprojekt p 
INNER JOIN	isum.v_bi_UmsWFHistorieStavu hist ON hist.nositelid=p.id
INNER JOIN	isum.v_bi_UmzVyzva v ON v.id=p.id_vyzva
 
WHERE   p.KODWFSTAV in ('PP21','PP22')
AND	hist.StavDo in ('PP21','PP22')
AND	hist.StavZ <> 'PP23a' AND hist.StavZ <> 'PP23b'
AND	p.kodprioritniosa = '01.3'
-- AND	p."Registraèní èíslo projektu" = 'CZ.01.3.14/0.0/0.0/19_252/0019855'
