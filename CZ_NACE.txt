	"ISUM"."V_BI_UMSRSCZNACE" "V_BI_UMSRSCZNACE",

	and	"V_BI_UMZVYZVA"."ID"="V_BI_UMZPROJEKT"."ID_VYZVA"

		p."Název projektu CZ",
