SELECT 
B."Èíslo výzvy",
B."Název výzvy",
B."Kód",
B."Název",
B."Check VP po staru"


from
(SELECT 
v."Èíslo výzvy" as "Èíslo výzvy 1",
i."Název" as "Název hodnoty z èíselníku",
i."ID"

FROM isum.V_BI_UMZVYZVA v
inner join isum.V_BI_UMZVYZVAMD d on d.ID_VYZVA=v.id
inner join isum.V_BI_UMSCISZAKLADNI i on i.id=d.ID_CISZAKLADNI
inner join isum.V_BI_UMSCISSEZNAM m on m.id=d.ID_CISSEZNAM

WHERE m."Název"='Skupina rozpoètu'
and d."Pro plnou žádost"='A'
and d."Platnost"='A'
and d."ID_CISSEZNAM" <>'150032') A

inner join

(select	 "V_BI_UMZVYZVA"."Èíslo výzvy" as "Èíslo výzvy",
	 "V_BI_UMZVYZVA"."Název výzvy" as "Název výzvy",
	 "V_BI_UMZVYZVA"."PREDPOMERYPROVP" as "Check VP po staru",
	 "V_BI_UMFROZPOLOZKY"."Kód" as "Kód",
	 "V_BI_UMFROZPOLOZKY"."Název" as "Název",
	"V_BI_UMFROZHLAVICKA"."ID_SKUPINA",
	 "V_BI_UMFROZHLAVICKA"."Název" as "Název_1" 

 from	"ISUM"."V_BI_UMFROZPOLOZKY" "V_BI_UMFROZPOLOZKY",
	"ISUM"."V_BI_UMFROZHLAVICKA" "V_BI_UMFROZHLAVICKA",
	"ISUM"."V_BI_UMZPROJEKT" "V_BI_UMZPROJEKT",
	"ISUM"."V_BI_UMZVYZVA" "V_BI_UMZVYZVA" 

 where   "V_BI_UMZPROJEKT"."ID_VYZVA"="V_BI_UMZVYZVA"."ID"
  and	 "V_BI_UMFROZHLAVICKA"."ID_NROZPOCET"="V_BI_UMZPROJEKT"."ID"
  and	 "V_BI_UMFROZPOLOZKY"."ID_HLAVICKA"="V_BI_UMFROZHLAVICKA"."ID"
  and	 "V_BI_UMFROZPOLOZKY"."LISTOVA" ='A' 
  and 	"V_BI_UMFROZPOLOZKY"."Název" not like '%minimis%' 
  and 	"V_BI_UMFROZPOLOZKY"."Název" not like '%MINIMIS%' ) B

on A."ID"=B."ID_SKUPINA"
