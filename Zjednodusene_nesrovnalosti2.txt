select	 nesr."Èíslo pøípadu",
	 nesr."Název pøíjemce",
	 nesr."Název koneèného pøíjemce",
  	 nesr."TYP",
	 hs."STAVDO",
--	 to_char(min(hs."CREATE_CAS"),'dd.mm.yyyy') as CasPrepnuti,
	 min(hs."CREATE_CAS") as CasPrepnuti,
	 hs."CREATE_UZIVATEL"

from	isum.v_bi_umznesrovnalost nesr
	inner join isum.v_bi_umswfhistoriestavu hs on nesr.id=hs.nositelid

where   
-- PØÍPAD NÍŽE JE PRO TESTOVACÍ ÚÈELY ZOBRAZENÉHO VÝSTUPU	
 	nesr."Èíslo pøípadu" in ('CZ.01.2.06/0.0/0.0/15_038/0006617/18/001','CZ.01.1.02/0.0/0.0/15_002/0000344/18/001') and	 

	(hs."STAVDO" ='SCHV' or	hs."STAVDO" ='SCHVAL') and

-- ODSTRANÌNÍ ZÁZNAMÙ SCHÁLENÝCH PANÍ Joachimsthalerovou z OP D 
	hs."CREATE_UZIVATEL" <> 'JOAHEL' and
	nesr."Èíslo pøípadu" is not null

group by nesr."Èíslo pøípadu", 
	nesr."Název pøíjemce", 
	hs."STAVDO" ,
	nesr."TYP",	 
 	nesr."Název koneèného pøíjemce",
	hs."CREATE_UZIVATEL"

order by nesr."Èíslo pøípadu"
