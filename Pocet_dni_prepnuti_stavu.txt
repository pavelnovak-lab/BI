SELECT 	p."Registraèní èíslo projektu",
	v."Èíslo výzvy",
	p.kodwfstav,
	hs.datumod,
	(select count(*) 
		from isum.V_BI_UMSKALENDAR r 
		where r.datum between trunc(hs.datumod) and trunc(sysdate) 
		and r.dentydne not in (6,7) 
		and r.SVATEK='N') AS "Poèet dní po FNaP"


FROM 		isum.v_bi_Umzprjggii p
INNER JOIN	isum.v_bi_umswfhistoriestavu hs ON hs.nositelid=p.id 
INNER JOIN	isum.v_bi_umzvyzva v ON p.id_vyzva=v.id



WHERE 	
-- v."Èíslo výzvy"='01_19_252'
-- AND	
p."Registraèní èíslo projektu" like ('CZ.01.3%')
AND hs.STAVDO in ('PP21','PP22')
AND	p.kodwfstav in ('PP21','PP22')
/* AND NOT EXISTS
(SELECT h.id_druhhodn FROM isum.v_bi_Umzprjggii p INNER JOIN isum.v_bi_umvhodnotitelkompprj h ON p.id=h.id_noshodnoceni WHERE h.id_druhhodn=171152)
*/
ORDER BY "Poèet dní po FNaP"
